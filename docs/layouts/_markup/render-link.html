{#- Resolves a relative destination to a page, a page resource or an asset, like the embedded hook, and warns when an internal one resolves to nothing or names a fragment the page lacks: the docs build test fails on warnings. The warnings have no id, because warnings with an id are reported once. External links get `rel="external"`. -#} {%- set u = destination | parse_url -%} {%- set href = u.string -%} {%- set external = u.is_absolute -%} {%- if href is starting_with(pat="#") -%} {%- set href = page_inner.rel_permalink ~ "#" ~ u.fragment -%} {%- elif href and not external and not (href is starting_with(pat="mailto:")) -%} {%- set path = u.path | strip_prefix(prefix="./") -%} {%- if path -%} {%- set target = get_page(path=path, page=page_inner) or (page_inner.resources | get_resource(name=path)) or get_asset(path=path) -%} {%- if target -%} {%- set href = target.rel_permalink -%} {%- if u.query %}{% set href = href ~ "?" ~ u.query %}{% endif -%} {%- if u.fragment -%} {%- set href = href ~ "#" ~ u.fragment -%} {%- if target.kind is defined -%} {%- set fragments = page_fragments(page=target) -%} {%- if not (u.fragment in fragments.identifiers) -%} {{- log_warn(message="broken fragment " ~ destination ~ " in " ~ page_inner.path) -}} {%- endif -%} {%- endif -%} {%- endif -%} {%- else -%} {{- log_warn(message="broken link " ~ destination ~ " in " ~ page_inner.path) -}} {%- endif -%} {%- endif -%} {%- endif -%} {{ text }} {#- -#}
