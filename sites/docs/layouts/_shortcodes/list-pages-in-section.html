{#- Renders a description list of the pages in the given section. Render a subset of the pages in the section by specifying a predefined filter, and whether to include those pages. See " ~ shortcode.position) }} {% endif %} {% endfor %} {% else %} {{ log_error(message="The \"" ~ shortcode.name ~ "\" shortcode was unable to find the \"" ~ filter ~ "\" filter in the page_filters data file. See " ~ shortcode.position) }} {% endif %} {% endif %} {# Render. #} {% set section_path = shortcode | arg(name="path", default="") %} {% if section_path %} {% set section = get_page(path=section_path) %} {% if section %} {% if section.regular_pages %} {% for p in section.regular_pages | by_title %} {% if (filter_type == "include" and p.id in filtered_ids) or (filter_type == "exclude" and p.id not in filtered_ids) or filter_type == "none" %} {% set link_title = title_prefix ~ p.link_title %} [{{ link_title }}]({{ p.rel_permalink }}){# Do not indent. #} : {{ p.description }}{# Do not indent. #} {% endif %} {% endfor %} {% else %} {{ log_warn(message="The \"" ~ shortcode.name ~ "\" shortcode found no pages in the \"" ~ section_path ~ "\" section. See " ~ shortcode.position) }} {% endif %} {% else %} {{ log_error(message="The \"" ~ shortcode.name ~ "\" shortcode was unable to find \"" ~ section_path ~ "\". See " ~ shortcode.position) }} {% endif %} {% else %} {{ log_error(message="The \"" ~ shortcode.name ~ "\" shortcode requires a 'path' parameter indicating the path to the section.
