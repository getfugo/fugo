{#- Renders syntax-highlighted configuration data in JSON, TOML, and YAML formats. @param {string} [config] The section of site.data.docs.config to render. @param {bool} [copy=false] Whether to display a copy-to-clipboard button. @param {string} [dataKey] The section of site.data.docs to render. @param {string} [file] The file name to display above the rendered code. @param {bool} [fm=false] Whether to render the code as front matter. @param {bool} [skipHeader=false] Whether to omit top level key(s) when rendering a section of site.data.docs.config. @example {{< code-toggle file=config config=build />}} @example {{< code-toggle file=content/example.md fm="true" }} title='Example' draft='false {{< /code-toggle }} Go wrapped a config section in its keys with `dict $sections $section`; here each key wraps it, innermost first, with `[[key, value]] | from_pairs` (map literal keys are constants). The i01 variant (sites.py DOCS_REPLACE): no highlighting (the code is escaped into a plain 
), and every tab shows JSON (remarshal → "json").
-#}
{#- Get parameters. #}
{%- set config = shortcode | arg(name="config", default="") %}
{%- set data_key = shortcode | arg(name="dataKey", default="") %}
{%- set file = shortcode | arg(name="file", default="") %}
{%- set copy = (shortcode | arg(name="copy", default="")) in ["true", true, 1] %}
{%- set fm = (shortcode | arg(name="fm", default="")) in ["true", true, 1] %}
{%- set skip_header = (shortcode | arg(name="skipHeader", default="")) in ["true", true, 1] %}

{#- Define constants. #}
{%- set delimiters = {"toml": "+++", "yaml": "---"} %}
{%- set langs = ["yaml", "toml", "json"] %}
{%- set place_holder = "#-hugo-placeholder-#" %}

{#- Render. #}
{%- if config %}
  {%- set file = file or "hugo" %}
  {%- set sections = config | split(pat=".") %}
  {%- set config_section = site.data.docs.config | get_path(path=sections) %}
  {%- if skip_header %}
    {%- set code = config_section %}
  {%- else %}
    {%- set_global code = config_section %}
    {%- for s in sections | reverse %}
      {%- set_global code = [[s, code]] | from_pairs %}
    {%- endfor %}
  {%- endif %}
{%- elif data_key %}
  {%- set file = file or data_key %}
  {%- set code = site.data.docs | get_path(path=data_key | split(pat=".")) %}
{%- else %}
  {%- set code = inner %}
{%- endif %}

    {%- endif %}

  {%- if code %}
    {%- for l in langs %}

          {%- set h_code = place_holder ~ "\n" ~ h_code ~ "\n" ~ place_holder %}
